cmake_minimum_required(VERSION 3.20)
project(mimir_runtime LANGUAGES C CXX)
set(LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../llama.cpp" CACHE PATH "Patched llama.cpp source")
option(MIMIR_BUILD_TESTS "Build runtime integration tests" ON)
option(MIMIR_BUILD_CHAT "Build text chat library and terminal client" ON)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ${MIMIR_BUILD_CHAT} CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
file(READ "${LLAMA_SOURCE}/include/llama.h" LLAMA_HEADER)
if(NOT LLAMA_HEADER MATCHES "llama_decode_prefix")
    message(FATAL_ERROR "Apply native/mimir/patches/prefixlm-engine.patch to llama.cpp first")
endif()
add_subdirectory("${LLAMA_SOURCE}" llama)
add_library(mimir-runtime STATIC session.cpp)
target_compile_features(mimir-runtime PUBLIC cxx_std_17)
target_include_directories(mimir-runtime PUBLIC include)
target_link_libraries(mimir-runtime PUBLIC llama)
target_compile_options(mimir-runtime PRIVATE -Wall -Wextra -Wpedantic)
if(MIMIR_BUILD_CHAT)
    file(READ "${LLAMA_SOURCE}/src/llama-vocab.h" LLAMA_VOCAB_HEADER)
    if(NOT LLAMA_VOCAB_HEADER MATCHES "LLAMA_VOCAB_PRE_TYPE_SPM_BPE_MISTRAL")
        message(FATAL_ERROR "Apply native/mimir/patches/text-codec.patch before building chat")
    endif()
    add_library(mimir-chat-lib STATIC text.cpp chat.cpp)
    target_compile_features(mimir-chat-lib PUBLIC cxx_std_17)
    target_include_directories(mimir-chat-lib PUBLIC include)
    target_link_libraries(mimir-chat-lib PUBLIC mimir-runtime PRIVATE llama-common)
    target_compile_options(mimir-chat-lib PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(mimir-chat tools/chat.cpp)
    target_link_libraries(mimir-chat PRIVATE mimir-chat-lib)
    target_include_directories(mimir-chat PRIVATE "${LLAMA_SOURCE}/vendor")
    set_target_properties(mimir-chat PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
if(MIMIR_BUILD_TESTS)
    enable_testing()
    if(MIMIR_BUILD_CHAT)
        add_executable(mimir-text-tests tests/text.cpp)
        target_link_libraries(mimir-text-tests PRIVATE mimir-chat-lib llama-common)
        set_target_properties(mimir-text-tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
        add_test(NAME text-unit COMMAND mimir-text-tests)
        add_executable(mimir-jinja-tests "${LLAMA_SOURCE}/tests/test-jinja.cpp")
        target_include_directories(mimir-jinja-tests PRIVATE "${LLAMA_SOURCE}/tests")
        target_link_libraries(mimir-jinja-tests PRIVATE llama-common)
        add_test(NAME jinja-regression COMMAND mimir-jinja-tests "string|filters")
        add_executable(mimir-reasoning-tests "${LLAMA_SOURCE}/tests/test-reasoning-budget.cpp")
        target_link_libraries(mimir-reasoning-tests PRIVATE llama-common)
        add_test(NAME reasoning-persistence COMMAND mimir-reasoning-tests)
        add_executable(mimir-backend-sampler-tests "${LLAMA_SOURCE}/tests/test-backend-sampler.cpp")
        target_link_libraries(mimir-backend-sampler-tests PRIVATE llama-common)
        add_executable(mimir-engine-tests "${LLAMA_SOURCE}/tests/test-llama-archs.cpp")
        target_link_libraries(mimir-engine-tests PRIVATE llama-common)
        foreach(ARCH hrm_text llama)
            add_test(NAME prefix-engine-${ARCH} COMMAND mimir-engine-tests --prefix-lm --arch ${ARCH} --seed 42)
            set_tests_properties(prefix-engine-${ARCH} PROPERTIES TIMEOUT 300)
        endforeach()
    endif()
    add_executable(mimir-session-tests tests/session.cpp)
    target_include_directories(mimir-session-tests PRIVATE "${LLAMA_SOURCE}/vendor")
    target_link_libraries(mimir-session-tests PRIVATE mimir-runtime)
    set_target_properties(mimir-session-tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    set(MIMIR_FIXTURE "${CMAKE_CURRENT_SOURCE_DIR}/../../logs/prefixlm-comparison/fixture" CACHE PATH "Comparison fixture")
    foreach(METADATA true false absent)
        add_test(NAME session-cpu-${METADATA} COMMAND mimir-session-tests
            "${MIMIR_FIXTURE}/hrm-${METADATA}.gguf" "${MIMIR_FIXTURE}" cpu off
            "${CMAKE_BINARY_DIR}/results/cpu-${METADATA}")
        set_tests_properties(session-cpu-${METADATA} PROPERTIES TIMEOUT 300)
    endforeach()
endif()
