cmake_minimum_required(VERSION 3.20)
project(prefixlm_comparison LANGUAGES C CXX)
if(NOT LLAMA_SOURCE)
    message(FATAL_ERROR "Set LLAMA_SOURCE to the pinned llama.cpp worktree")
endif()
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama)
add_executable(prefixlm-runner runner.cpp)
target_compile_features(prefixlm-runner PRIVATE cxx_std_17)
target_include_directories(prefixlm-runner PRIVATE "${LLAMA_SOURCE}/vendor")
target_link_libraries(prefixlm-runner PRIVATE llama)
file(READ "${LLAMA_SOURCE}/include/llama.h" LLAMA_HEADER)
string(FIND "${LLAMA_HEADER}" "llama_model_is_prefix_lm" PREFIXLM_QUERY_POSITION)
if(NOT PREFIXLM_QUERY_POSITION EQUAL -1)
    target_compile_definitions(prefixlm-runner PRIVATE PREFIXLM_HAS_QUERY)
endif()
set_target_properties(prefixlm-runner PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
